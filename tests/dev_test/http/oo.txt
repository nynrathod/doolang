import std::Http::{Server, Request, Response, Next};

enum AuthError {
    Unauthorized,
    Forbidden,
}

struct ApiResponse {
    Message: Str,
    UserId: Int,
}

fn AuthMiddleware(req: Request, next: Next) -> Response ! AuthError {
    print("🔍 AuthMiddleware: START");
    let token = req.header("Authorization");
    print("🔍 AuthMiddleware: Got token:", token);

    if token == "" {
        print("❌ AuthMiddleware: No token provided - returning Unauthorized");
        return Err AuthError::Unauthorized;
    }

    if token != "Bearer valid-token" {
        print("❌ AuthMiddleware: Invalid token - returning Unauthorized");
        return Err AuthError::Unauthorized;
    }

    print("✓ AuthMiddleware: Token valid, calling next...");
    let response = next.call();
    print("✓ AuthMiddleware: Got response from next, returning Ok");
    return Ok response;
}

fn AdminMiddleware(req: Request, next: Next) -> Response ! AuthError {
    print("🔍 AdminMiddleware: START");
    let role = req.header("X-Role");
    print("🔍 AdminMiddleware: Got role:", role);

    if role != "admin" {
        print("❌ AdminMiddleware: Not admin role - returning Forbidden");
        return Err AuthError::Forbidden;
    }

    print("✓ AdminMiddleware: Admin role confirmed, calling next...");
    let response = next.call();
    print("✓ AdminMiddleware: Got response from next, returning Ok");
    return Ok response;
}

fn LogMiddleware(req: Request, next: Next) -> Response {
    print("→ " + req.Method + " " + req.Path);
    let res = next.call();
    print("← Status: " + res.Status);
    return res;
}

fn GetPublic() -> ApiResponse {
    print("📍 Handler: GetPublic");
    return ApiResponse {
        Message: "Public data",
        UserId: 0,
    };
}

fn GetProfile() -> ApiResponse {

    };
}

fn GetAdmin() -> ApiResponse {

    return ApiResponse {

}

// 13_field_decorators_test.doo
// Test @writeOnly, @readOnly, @internal decorators with HTTP and Database

import std::Http::Server;
import std::Database;

// User model with field visibility decorators
struct User {
    id: Int @primary @auto,
    Email: Str @email @unique,
    Password: Str @hash @writeOnly,          // In request, NOT in response
    Name: Str,
    Credits: Int @readOnly @default(100),     // NOT in request, in response
    Role: Str @default("user"),
    InternalId: Str @internal,               // NOT in request, NOT in response (backend only)
}

// Response struct for user profile
struct UserProfile {
    id: Int,
    email: Str,
    name: Str,
    credits: Int,
    role: Str,
}

// Convert User to UserProfile (manual mapping for response)
fn ToProfile(user: User) -> UserProfile {
    return UserProfile {
        id: user.id,
        email: user.Email,
        name: user.Name,
        credits: user.Credits,
        role: user.Role,
    };
}

// Test handler: Get all users
fn GetUsers() -> [UserProfile] ! DatabaseError {
    let db = Database::get()?;
    let users: [User] = db.raw("SELECT * FROM users")?;
    let mut profiles: [UserProfile] = [];

    for user in users {
        profiles.push(ToProfile(user));
    }

    Ok profiles;
}

// Test handler: Get single user
fn GetUser(id: Int) -> UserProfile ! DatabaseError {
    let db = Database::get()?;

    let user: User = db.rawWithParams("SELECT * FROM users WHERE id = $1", id)?;
    Ok ToProfile(user);
}
static DB: Database;
fn main() {
    DB = Database::Postgres()?;
    let app = Server::new(":3113");

    // Auth endpoints - will auto-handle @writeOnly password
    app.auth("/signup", "/login", User, DB);

    // CRUD - will respect decorators
    app.crud("/users", User, DB);

    // Custom endpoints
    app.get("/users-all", GetUsers);

    app.start();
}

import std::Database;

enum UserRole {
    User,
    Admin,
    Moderator,
}

enum Status { Todo, InProgress, Done }
enum Priority { Low, Medium, High }


struct UserInput {
    Email: Str @email,                       // Must be valid email
    Age: Int @min(18),                       // Must be Int, >= 18
    Score: Int @max(100),                    // Must be Int, <= 100
    Username: Str @min(3) @max(50),          // Must be Str, length between 3 and 50
    Rank: Int @min(1) @max(5),               // Must be Int, between 1 and 5
    Role: UserRole,   // Must be one of these values
    Name: Str,                               // String field
}

struct User {
    id: Int @primary @auto,
    email: Str @email @unique,
    password: Str @hash @min(6),
}

struct Task {
    id: Int @primary @auto,
    title: Str @min(1) @max(200),
    status: Status @default(Status::Todo),
    priority: Priority @default(Priority::Medium),
    userId: Int @foreign(User),
}

fn createUser(input: UserInput) -> UserInput {
    return UserInput {
        Email: input.Email,
        Age: input.Age,
        Score: input.Score,
        Username: input.Username,
        Rank: input.Rank,
        Role: input.Role,
        Name: input.Name
    };
}

fn main() {
    let db = Database::Postgres()?;
    let app = Server::new(":3104");

    app.auth("/signup", "/login", User, db);
    app.post("/api/users/create", createUser);
    app.crud("/tasks", Task, db);

    app.start();
}
